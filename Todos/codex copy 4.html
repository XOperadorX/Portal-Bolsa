🔧 Criar Função (Executar 1x)

-- ==================================================
-- FUNÇÃO: registrar_item_codex
-- Tabela alvo: public."Geral"
-- Executar UMA VEZ apenas
-- ==================================================
CREATE OR REPLACE FUNCTION public.registrar_item_codex(
  p_login          VARCHAR(255),
  p_id_item        TEXT,
  p_nome           TEXT,
  p_tipo_peca      TEXT,
  p_raridade       TEXT,
  p_tier           INTEGER,
  p_id_conjunto    TEXT,
  p_nome_conjunto  TEXT,
  p_bonus_ataque   INTEGER,
  p_bonus_defesa   INTEGER,
  p_bonus_hp       INTEGER
)
RETURNS TEXT AS $$
DECLARE
  v_item_json        JSONB;
  v_existe           BOOLEAN;
  v_total_pecas      INTEGER;
  v_pecas_registradas INTEGER;
  v_bonus_atk_item   INTEGER;
  v_bonus_def_item   INTEGER;
  v_bonus_hp_item    INTEGER;
  v_bonus_atk_conj   INTEGER := 0;
  v_bonus_def_conj   INTEGER := 0;
  v_bonus_hp_conj    INTEGER := 0;
  v_retorno          TEXT;
BEGIN
  -- Verificar usuário
  SELECT EXISTS(SELECT 1 FROM public."Geral" WHERE login = p_login) INTO v_existe;
  IF NOT v_existe THEN
    RETURN 'ERRO: Usuário não encontrado: ' || p_login;
  END IF;

  -- Verificar item duplicado
  SELECT EXISTS(
    SELECT 1 FROM public."Geral", jsonb_array_elements(itens) AS i
    WHERE login = p_login AND i->>'id_item' = p_id_item
  ) INTO v_existe;
  IF v_existe THEN
    RETURN 'AVISO: Item já registrado — ' || p_nome;
  END IF;

  -- Definir total de peças por tipo de conjunto
  v_total_pecas := CASE 
    WHEN p_id_conjunto LIKE '%acess%' THEN 5
    ELSE 7
  END;

  -- Calcular valores individuais
  v_bonus_atk_item := p_bonus_ataque;
  v_bonus_def_item := p_bonus_defesa;
  v_bonus_hp_item  := p_bonus_hp;

  -- Montar JSON do item
  v_item_json = jsonb_build_object(
    'id_item', p_id_item,
    'nome', p_nome,
    'tipo_peca', p_tipo_peca,
    'raridade', p_raridade,
    'tier', p_tier,
    'registrado_em', NOW()::TEXT,
    'bônus', jsonb_build_object(
      'ataque', v_bonus_atk_item,
      'defesa', v_bonus_def_item,
      'hp', v_bonus_hp_item
    ),
    'conjunto', jsonb_build_object(
      'id_conjunto', p_id_conjunto,
      'nome_conjunto', p_nome_conjunto,
      'completo', false
    ),
    'pode_descartar', true
  );

  -- Inserir item
  UPDATE public."Geral"
  SET itens = itens || v_item_json
  WHERE login = p_login;

  -- Contar peças do conjunto
  SELECT COUNT(*) INTO v_pecas_registradas
  FROM public."Geral", jsonb_array_elements(itens) AS i
  WHERE login = p_login AND i->'conjunto'->>'id_conjunto' = p_id_conjunto;

  -- Verificar conjunto completo
  IF v_pecas_registradas = v_total_pecas THEN
    -- Bônus extra de conjunto completo
    v_bonus_atk_conj := CASE 
      WHEN p_raridade = 'lendario' THEN 60
      WHEN p_raridade = 'epico' THEN 40
      WHEN p_raridade = 'raro' THEN 25
      ELSE 10
    END;
    v_bonus_def_conj := CASE 
      WHEN p_raridade = 'lendario' THEN 40
      WHEN p_raridade = 'epico' THEN 27
      WHEN p_raridade = 'raro' THEN 17
      ELSE 7
    END;
    v_bonus_hp_conj := CASE 
      WHEN p_raridade = 'lendario' THEN 800
      WHEN p_raridade = 'epico' THEN 500
      WHEN p_raridade = 'raro' THEN 300
      ELSE 120
    END;

    -- Marcar todas as peças do conjunto como completo
    UPDATE public."Geral" u
    SET itens = jsonb_set(
      itens,
      ARRAY[(idx-1)::TEXT, 'conjunto'],
      jsonb_build_object(
        'id_conjunto', p_id_conjunto,
        'nome_conjunto', p_nome_conjunto,
        'completo', true
      )
    )
    FROM generate_subscripts(u.itens, 1) AS idx
    WHERE u.login = p_login
      AND u.itens->(idx-1)->'conjunto'->>'id_conjunto' = p_id_conjunto;

    v_retorno := '✅ SUCESSO: ' || p_nome || ' registrado! CONJUNTO COMPLETO: ' || p_nome_conjunto || 
                 ' (+' || v_bonus_atk_conj || ' ATK, +' || v_bonus_def_conj || ' DEF, +' || v_bonus_hp_conj || ' HP)';
  ELSE
    v_retorno := '✅ SUCESSO: ' || p_nome || ' registrado! Progresso: ' || v_pecas_registradas || '/' || v_total_pecas;
  END IF;

  -- Recalcular atributos finais
  WITH soma_bonus AS (
    SELECT
      -- Bônus de itens individuais
      COALESCE(SUM((i->'bônus'->>'ataque')::integer), 0) AS atk_item,
      COALESCE(SUM((i->'bônus'->>'defesa')::integer), 0) AS def_item,
      COALESCE(SUM((i->'bônus'->>'hp')::integer), 0) AS hp_item,
      -- Bônus de conjuntos completos
      (SELECT COALESCE(SUM(CASE 
         WHEN c.raridade = 'lendario' THEN 60
         WHEN c.raridade = 'epico' THEN 40
         WHEN c.raridade = 'raro' THEN 25
         ELSE 10 END), 0)
       FROM (SELECT DISTINCT ON (i2->'conjunto'->>'id_conjunto') 
             i2->'conjunto'->>'id_conjunto' AS cid,
             i2->>'raridade' AS raridade
             FROM public."Geral" u2, jsonb_array_elements(u2.itens) AS i2
             WHERE u2.login = p_login 
               AND i2->'conjunto'->>'completo' = 'true') AS c
      ) AS atk_conj,
      (SELECT COALESCE(SUM(CASE 
         WHEN c.raridade = 'lendario' THEN 40
         WHEN c.raridade = 'epico' THEN 27
         WHEN c.raridade = 'raro' THEN 17
         ELSE 7 END), 0)
       FROM (SELECT DISTINCT ON (i2->'conjunto'->>'id_conjunto') 
             i2->'conjunto'->>'id_conjunto' AS cid,
             i2->>'raridade' AS raridade
             FROM public."Geral" u2, jsonb_array_elements(u2.itens) AS i2
             WHERE u2.login = p_login 
               AND i2->'conjunto'->>'completo' = 'true') AS c
      ) AS def_conj,
      (SELECT COALESCE(SUM(CASE 
         WHEN c.raridade = 'lendario' THEN 800
         WHEN c.raridade = 'epico' THEN 500
         WHEN c.raridade = 'raro' THEN 300
         ELSE 120 END), 0)
       FROM (SELECT DISTINCT ON (i2->'conjunto'->>'id_conjunto') 
             i2->'conjunto'->>'id_conjunto' AS cid,
             i2->>'raridade' AS raridade
             FROM public."Geral" u2, jsonb_array_elements(u2.itens) AS i2
             WHERE u2.login = p_login 
               AND i2->'conjunto'->>'completo' = 'true') AS c
      ) AS hp_conj
    FROM public."Geral", jsonb_array_elements(itens) AS i
    WHERE login = p_login
  )
  UPDATE public."Geral"
  SET
    atk    = ataque_base + atk_item + atk_conj,
    def    = defesa_base + def_item + def_conj,
    hp_max = hp_max      + hp_item  + hp_conj
  FROM soma_bonus
  WHERE login = p_login;

  RETURN v_retorno;
END;
$$ LANGUAGE plpgsql;
