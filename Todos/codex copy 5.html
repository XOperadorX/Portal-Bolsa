🔧 Função de Registro

Execute no PostgreSQL para criar a função de registro automático:

-- ==================================================
-- FUNÇÃO: registrar_item_codex
-- ==================================================
CREATE OR REPLACE FUNCTION public.registrar_item_codex(
  p_login          VARCHAR(255),
  p_id_item        TEXT,
  p_nome           TEXT,
  p_tipo_peca      TEXT,
  p_raridade       TEXT,
  p_tier           INTEGER,
  p_id_conjunto    TEXT,
  p_nome_conjunto  TEXT,
  p_bonus_ataque   INTEGER,
  p_bonus_defesa   INTEGER,
  p_bonus_hp       INTEGER
)
RETURNS TEXT AS $$
DECLARE
  v_item_json JSONB;
  v_existe BOOLEAN;
  v_total_pecas INTEGER;
  v_pecas_registradas INTEGER;
  v_retorno TEXT;
BEGIN
  SELECT EXISTS(SELECT 1 FROM public."Geral" WHERE login = p_login) INTO v_existe;
  IF NOT v_existe THEN
    RETURN 'ERRO: Usuário não encontrado: ' || p_login;
  END IF;

  SELECT EXISTS(
    SELECT 1 FROM public."Geral", jsonb_array_elements(itens) AS i
    WHERE login = p_login AND i->>'id_item' = p_id_item
  ) INTO v_existe;
  IF v_existe THEN
    RETURN 'AVISO: Item já registrado — ' || p_nome;
  END IF;

  v_total_pecas := CASE WHEN p_id_conjunto LIKE '%acess%' THEN 5 ELSE 7 END;

  v_item_json = jsonb_build_object(
    'id_item', p_id_item, 'nome', p_nome, 'tipo_peca', p_tipo_peca,
    'raridade', p_raridade, 'tier', p_tier,
    'registrado_em', NOW()::TEXT,
    'bônus', jsonb_build_object(
      'ataque', p_bonus_ataque, 'defesa', p_bonus_defesa, 'hp', p_bonus_hp
    ),
    'conjunto', jsonb_build_object(
      'id_conjunto', p_id_conjunto, 'nome_conjunto', p_nome_conjunto, 'completo', false
    ),
    'pode_descartar', true
  );

  UPDATE public."Geral" SET itens = itens || v_item_json WHERE login = p_login;

  SELECT COUNT(*) INTO v_pecas_registradas
  FROM public."Geral", jsonb_array_elements(itens) AS i
  WHERE login = p_login AND i->'conjunto'->>'id_conjunto' = p_id_conjunto;

  IF v_pecas_registradas = v_total_pecas THEN
    UPDATE public."Geral" u
    SET itens = jsonb_set(itens, ARRAY[(idx-1)::TEXT, 'conjunto'],
      jsonb_build_object('id_conjunto', p_id_conjunto, 'nome_conjunto', p_nome_conjunto, 'completo', true))
    FROM generate_subscripts(u.itens, 1) AS idx
    WHERE u.login = p_login AND u.itens->(idx-1)->'conjunto'->>'id_conjunto' = p_id_conjunto;

    v_retorno := '✅ SUCESSO: ' || p_nome || ' registrado! CONJUNTO COMPLETO!';
  ELSE
    v_retorno := '✅ SUCESSO: ' || p_nome || ' registrado! Progresso: ' || v_pecas_registradas || '/' || v_total_pecas;
  END IF;

  RETURN v_retorno;
END;
$$ LANGUAGE plpgsql;
